import { AutoroutingPipelineSolver9_PreloadedTraceGraph,
  type SimpleRouteJson, type Obstacle } from "@tscircuit/capacity-autorouter"

const layers = ["top", "inner1", "inner2", "inner3", "inner4", "bottom"]

// Freeze the actual copper from completed phases. Running graph repair on
// preloaded routes in capacity-autorouter 0.0.951 produced an invalid layer
// jump in the byte-lane experiment. A fresh Pipeline 9 solver still performs
// each autorouting phase, with earlier copper as immutable obstacles.
function fixedCopper(traces: NonNullable<SimpleRouteJson["traces"]>): Obstacle[] {
  return traces.flatMap((trace,ti) => {
    const obstacles: Obstacle[] = []
    const connectedTo = [trace.connection_name,trace.pcb_trace_id,...(trace.connectsTo??[])]
      .filter((id):id is string=>typeof id === "string")
    for(let i=0;i<trace.route.length;i++) {
      const a=trace.route[i],b=trace.route[i+1]
      if(a.route_type === "via") {
        const diameter = a.via_diameter ?? 0.3
        obstacles.push({obstacleId:`memory_fixed_${ti}_via_${i}`,type:"rect",shape:"circle",
          layers,center:{x:a.x,y:a.y},width:diameter,height:diameter,connectedTo})
      }
      if(a.route_type !== "wire" || b?.route_type !== "wire" || a.layer !== b.layer) continue
      const dx=b.x-a.x,dy=b.y-a.y,length=Math.hypot(dx,dy)
      if(length<1e-9) continue
      const width=Math.max(a.width,b.width)
      obstacles.push({obstacleId:`memory_fixed_${ti}_segment_${i}`,type:"rect",layers:[a.layer],
        center:{x:(a.x+b.x)/2,y:(a.y+b.y)/2},width:length+width,height:width,
        ccwRotationDegrees:Math.atan2(dy,dx)*180/Math.PI,connectedTo})
    }
    return obstacles
  })
}

export async function routeMemoryAroundEarlierCopper(input: SimpleRouteJson) {
  if(input.layerCount !== 6 || input.allowBlindAndBuriedVias !== false)
    throw new Error("Memory experiment requires six layers and full-depth vias")
  const previous = input.traces ?? []
  const routingInput: SimpleRouteJson = {...input,traces:undefined,
    obstacles:[...input.obstacles,...fixedCopper(previous)]}
  const solver = new AutoroutingPipelineSolver9_PreloadedTraceGraph(routingInput,{effort:2})
  const listeners:Record<string,((event:any)=>void)[]> = {}
  let output=previous,stopped=false
  const emit=(type:string,event:any)=>listeners[type]?.forEach(fn=>fn(event))
  const run=async()=>{
    try {
      let steps=0
      while(!stopped && !solver.solved && !solver.failed) {
        const start=Date.now(),iterations=solver.iterations
        while(!stopped && Date.now()-start<200 && !solver.solved && !solver.failed) solver.step()
        emit("progress",{steps:++steps,progress:solver.progress,phase:solver.getCurrentPhase(),
          iterationsPerSecond:(solver.iterations-iterations)*1000/Math.max(1,Date.now()-start)})
        await new Promise(resolve=>setTimeout(resolve,0))
      }
      if(stopped) return
      if(solver.failed) throw new Error(solver.error ?? "Memory autorouting failed")
      const traces=solver.getOutputSimplifiedPcbTraces()
      const covered=new Set(traces.map(trace=>trace.connection_name))
      for(const connection of input.connections)
        if(!covered.has(connection.name)) throw new Error(`Autorouter omitted ${connection.name}`)
      for(const trace of traces) {
        let lastLayer:string|undefined
        for(const point of trace.route) {
          if(point.route_type === "via") {
            if(lastLayer && point.from_layer !== lastLayer)
              throw new Error(`Disconnected via in ${trace.connection_name}`)
            lastLayer=point.to_layer
          } else if(point.route_type === "wire") {
            if(lastLayer && point.layer !== lastLayer)
              throw Object.assign(new Error(`Autorouter changed layers without a via in ${trace.connection_name}`),{invalidRoute:trace})
            lastLayer=point.layer
          } else throw new Error(`Unsupported ${point.route_type} route element in ${trace.connection_name}`)
        }
      }
      output=[...previous,...traces]
      emit("complete",{traces:output})
    } catch(error) {emit("error",{error,invalidRoute:(error as {invalidRoute?:unknown})?.invalidRoute})}
  }
  return {
    on(type:string,fn:(event:any)=>void){(listeners[type]??=[]).push(fn)},
    start(){void run()},
    stop(){stopped=true},
    getOutputSimpleRouteJson(){return {...input,traces:output}},
  }
}
